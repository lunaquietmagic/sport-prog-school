    import sqlite3

    # Подключение к файлу базы данных/его создание

    # con - connection/соединение с нашей базой данных

    with sqlite3.connect("project.db") as con:

    # Создание курсора, с помощью которого мы будем управлять базой данных

    cur = con.cursor()

    # Изменение данных в таблице

    cursor.execute("INSERT INTO clients (name, money) VALUES (?, ?)", (name_of_our_client,his_money_balance ))

    # cursor.execute - обращение к базе данных

    # INSERT INTO clients (name, money) - вносим нового клиента в таблицу clients и задаём ТОЛЬКО имя клиента и его баланс

    # VALUES (?, ?) - вносим данные из переменных, которые добавим следующим действием

    # (name_of_our_client,his_money_balance ) - а именно имя клиента из нашей программы и его баланс

    con.commit()

    # ОБЯЗАТЕЛЬНО применяем все изменения, а иначе данные не сохранятся в таблице (если мы просто получаем данные, то commit нам не нужен)
